\subsection{Contradiction}

\newcommand{\norbot}[1]{{\nor}#1{\nor}#1#1} % NOR hacking
\begin{notation}
    \[\bot \coloneqq P \nor (P \nor P) \qquad [\norbot{P}]\]
\end{notation}

\begin{foundation}
    \begin{equation}
        \inferrule{P \nor Q \\ P}{\bot} \quad \inferrule{P \nor Q \\ Q}{\bot}
        \tag{{$\nor$}E}
        \label{rule:nor-elimination}
    \end{equation}
    \begin{equation}
        \inferrule{[P] \cdots \bot \\ [Q] \cdots \bot}{P \nor Q}
        \tag{{$\nor$}I}
        \label{rule:nor-introduction}
    \end{equation}
    \begin{equation}
        \inferrule{[P \nor P] \cdots \bot}{P}
        \label{rule:nor-reductio-ad-absurdum}
    \end{equation}
\end{foundation}

\subsection{Negation}

\newcommand{\norneg}[1]{{\nor}#1#1} % NOR hacking
\begin{notation}
    \[\neg P \coloneqq P \nor P \qquad [\norneg{P}]\]
\end{notation}

\begin{align*}
    \premise{[P] \cdots \bot}\\
    \given{[P] \cdots \bot}\\
    \byrule{P \nor P}{\ref{rule:nor-introduction}}\\
    \bydefinition{\neg P}
\end{align*}
\begin{equation}
    \inferrule{[P] \cdots \bot}{\neg P}
    \tag{{$\neg$}I}
    \label{rule:negation-introduction}
\end{equation}

\begin{align*}
    \premise{\neg P}\\
    \premise{P}\\
    \bydefinition{P \nor P}\\
    \byrule{\bot}{\ref{rule:nor-elimination}}
\end{align*}

\begin{equation}
    \inferrule{\neg P \\ P}{\bot}
    \tag{{$\neg$}E}
    \label{rule:negation-elimination}
\end{equation}

\begin{align*}
    \premise{[\neg P] \cdots \bot}\\
    \bydefinition{[P \nor P] \cdots \bot}\\
    \byrule{P}{\ref{rule:nor-reductio-ad-absurdum}}
\end{align*}
\begin{equation}
    \inferrule{[\neg P] \cdots \bot}{P}
    \tag{RAA}
    \label{rule:reductio-ad-absurdum}
\end{equation}
