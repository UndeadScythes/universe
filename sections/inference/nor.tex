\newcommand{\norbot}[1]{{\nor}#1{\nor}#1#1}
\begin{notation}
    \[\bot \coloneqq P \nor (P \nor P) \qquad [\norbot{P}]\]
\end{notation}

\newcommand{\norE}{{\nor}E}
\newcommand{\norI}{{\nor}I}
\newcommand{\raaW}{RAA^{*}}
\begin{foundation}
    \[\inferrule{P \nor Q \\ P}{\bot} \quad \inferrule{P \nor Q \\ Q}{\bot} \quad (\norE)\]
    \[\inferrule{[P] \cdots \bot \\ [Q] \cdots \bot}{P \nor Q} \quad (\norI)\]
    \[\inferrule{[P \nor P] \cdots \bot}{P} \quad (\raaW)\]
\end{foundation}

\subsection{Negation}
\newcommand{\norneg}[1]{{\nor}#1#1}
\begin{notation}
    \[\neg P \coloneqq P \nor P \qquad [\norneg{P}]\]
\end{notation}

\begin{align*}
    \premise{[P] \cdots \bot}\\
    \reiterate{[P] \cdots \bot}\\
    \byrule{P \nor P}{\norI}\\
    \bydefinition{\neg P}
\end{align*}
\newcommand{\negI}{{\neg}I}
\[\inferrule{[P] \cdots \bot}{\neg P} \quad (\negI)\]

\begin{align*}
    \premise{\neg P}\\
    \premise{P}\\
    \bydefinition{P \nor P}\\
    \byrule{\bot}{\norE}
\end{align*}
\newcommand{\negE}{{\neg}E}
\[\inferrule{\neg P \\ P}{\bot} \quad (\negE)\]

\begin{align*}
    \premise{[\neg P] \cdots \bot}\\
    \bydefinition{[P \nor P] \cdots \bot}\\
    \byrule{P}{\raaW}
\end{align*}
\newcommand{\raa}{RAA}
\[\inferrule{[\neg P] \cdots \bot}{P} \quad (\raa)\]

% _ > P
\begin{align*}
    \premise{\bot}\\
    \assumption[1]{\neg P}\\
    \reiterate[1]{\bot}\\
    \byrule{P}{\raa}
\end{align*}
\newcommand{\botE}{{\bot}E}
\[\inferrule{\bot}{P} \quad (\botE)\]
