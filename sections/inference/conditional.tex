\subsection{Implication}

\newcommand{\norimplies}[2]{\norvee{\norneg{#1}}{#2}}
\begin{notation}
    \[P \implies Q \coloneqq \neg P \vee Q \qquad [\norimplies{P}{Q}]\]
\end{notation}

% [P]...Q > P=>Q
\begin{align*}
    \premise{[P] \cdots Q}\\
    \assumption[1]{\neg P \nor Q}\\
    \assumption[2]{P}\\
    \byrule[2]{Q}{\text{given}}\\
    \byrule[2]{\bot}{\norE}\\
    \byrule[1]{\neg P}{\negI}\\
    \byrule[1]{\bot}{\norE}\\
    \byrule{\neg(\neg P \nor Q)}{\negI}\\
    \bydefinition{\neg P \vee Q}\\
    \bydefinition{P \implies Q}
\end{align*}
\newcommand{\impliesI}{{\implies}I}
\[\inferrule{[P] \cdots Q}{P \implies Q} \quad (\impliesI)\]

% P, P=>Q > Q
\begin{align*}
    \premise{P}\\
    \premise{P \implies Q}\\
    \bydefinition{\neg(\neg P \nor Q)}\\
    \assumption[1]{\neg Q}\\
    \assumption[2]{\neg P}\\
    \byrule[2]{\bot}{\negE}\\
    \assumption[2]{Q}\\
    \byrule[2]{\bot}{\negE}\\
    \byrule[1]{\neg P \nor Q}{\norI}\\
    \byrule[1]{\bot}{\negE}\\
    \byrule{Q}{\raa}
\end{align*}
\newcommand{\impliesE}{{\implies}E}
\[\inferrule{P \\ P \implies Q}{Q} \quad (\impliesE)\]

% > P => P
\begin{align*}
    \assumption[1]{P}\\
    \reiterate[1]{P}\\
    \byrule{P \implies P}{\impliesI}    
\end{align*}
\newcommand{\impliesR}{{\implies}\text{Refl}}
\[\vdash P \implies P \quad (\impliesR)\]
